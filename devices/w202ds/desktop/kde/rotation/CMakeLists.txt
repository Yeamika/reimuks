cmake_minimum_required(VERSION 3.16)
project(w202_rotation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
find_package(KF6Screen 6.3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
add_executable(w202-rotation main.cpp)
target_link_libraries(w202-rotation PRIVATE Qt6::Widgets Qt6::DBus KF6::Screen PkgConfig::UDEV)
install(TARGETS w202-rotation DESTINATION bin)
install(FILES w202-rotation.desktop DESTINATION share/applications)
install(FILES w202-rotation-watch.desktop DESTINATION share/w202-rotation)
