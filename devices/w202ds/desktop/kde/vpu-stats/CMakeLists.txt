cmake_minimum_required(VERSION 3.16)
project(w202_vpu_stats LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(QT_MAJOR_VERSION 6)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(KF6CoreAddons REQUIRED)
find_package(KSysGuard 6.3 REQUIRED)
add_library(ksystemstats_plugin_w202_vpu MODULE plugin.cpp)
target_link_libraries(ksystemstats_plugin_w202_vpu PRIVATE Qt6::Core KF6::CoreAddons KSysGuard::SystemStats)
install(TARGETS ksystemstats_plugin_w202_vpu DESTINATION ${KDE_INSTALL_PLUGINDIR}/ksystemstats)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_gpu test_gpu.cpp)
    target_link_libraries(test_gpu PRIVATE Qt6::Core)
    add_test(NAME gpu_samples COMMAND test_gpu)
    add_executable(test_sample test_sample.cpp)
    add_test(NAME vpu_samples COMMAND test_sample)
    add_executable(test_plugin test_plugin.cpp)
    target_link_libraries(test_plugin PRIVATE Qt6::Core KF6::CoreAddons KSysGuard::SystemStats)
    add_test(NAME vpu_plugin_load COMMAND test_plugin $<TARGET_FILE:ksystemstats_plugin_w202_vpu>)
endif()
